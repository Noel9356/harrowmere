# The module's own build switch, declared here so everything about the module
# lives in its directory: the root build discovers this CMakeLists and the
# option arrives with it. OFF keeps every target defined - consumers still
# compile against the published capability header below, and an in-tree
# consumer that links the implementation still pulls it - but removes the
# plugin binary from the default build; a run whose mods then demand a
# capability only this module provides fails startup with a precise
# diagnostic naming the consumer, never a link error.
option(ENGINE_BUILD_MODULE_RADIANCE_CASCADES "Include the radiance_cascades module in the build" ON)

add_library(engine_module_radiance_cascades SHARED
    module_plugin.cpp
    radiance_cascades.cpp
)

target_link_libraries(engine_module_radiance_cascades
    PUBLIC engine_kernel
    PUBLIC engine_capability_logging
    PUBLIC engine_capability_renderer
    PUBLIC engine_capability_pbr
    PRIVATE engine_module_renderer_vulkan
    PRIVATE engine_vulkan_hpp_bindings
    # GLM behind the direction tables' vector math (module internal).
    PRIVATE engine_glm_bindings
)

if(ENGINE_BUILD_DEMO AND ENGINE_BUILD_MODULE_RADIANCE_CASCADES)
    include("${CMAKE_SOURCE_DIR}/cmake/shader_compilation.cmake")
    engine_compile_module_shaders(TARGET engine_module_radiance_cascades
                                  SOURCES shaders/rc_march.comp
                                          shaders/rc_compose.comp
                                          shaders/rc_gather.comp
                                          shaders/rc_denoise.comp
                                          shaders/rc_world_march.comp
                                          shaders/rc_world_compose.comp
                                          shaders/rc_world_gather.comp
                                  INCLUDE_DIRS shaders/common
                                            "${CMAKE_SOURCE_DIR}/modules/renderer_vulkan/shaders/common")
endif()

# The screen-space shaders (rc_march, rc_compose, rc_gather, rc_denoise) and
# the world-space shaders (rc_world_march, rc_world_compose, rc_world_gather)
# are all compiled into runtime/shaders/. The denoise shader rc_denoise.comp
# is SHARED between both placements - it operates on the gather output image
# which is screen-space regardless of where the probes lived. The placement
# config key "screen" (default) selects the screen-space pass; "world" selects
# the world-space pass; both load their shaders from the same
# runtime/shaders/ directory through the renderer's shader_directory.


# Module plugin deployment: the binary lands in runtime/modules/, the directory
# the kernel scans for plugins (see engine_kernel.cpp loadModulePlugins).
# PREFIX keeps one naming scheme on every platform (libengine_module_radiance_cascades.dll
# on Windows, libengine_module_radiance_cascades.so on POSIX); the $ORIGIN entries resolve
# the sibling plugins and $ORIGIN/.. the kernel and GLFW runtimes in runtime/;
# WINDOWS_EXPORT_ALL_SYMBOLS publishes the module's C++ surface to the other
# plugins that consume it across the boundary.
set_target_properties(engine_module_radiance_cascades PROPERTIES
    PREFIX "lib"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime/modules"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime/modules"
    BUILD_RPATH "${ENGINE_PLUGIN_RPATH_BUILD}"
    INSTALL_RPATH "${ENGINE_PLUGIN_RPATH_INSTALL}"
    WINDOWS_EXPORT_ALL_SYMBOLS ON
)
# MinGW: one explicit __declspec(dllexport) (the plugin entry point) disables
# the linker's all-symbol auto-export, and WINDOWS_EXPORT_ALL_SYMBOLS above is
# implemented only for MS-compatible toolchains; this flag publishes the
# module's C++ surface to the other plugins' import libraries (the lighting
# techniques reach RendererModule::attachLightingTechnique directly).
if(MINGW)
    target_link_options(engine_module_radiance_cascades PRIVATE -Wl,--export-all-symbols)
endif()


# This module provides demand marker capabilities (one per probe placement:
# "screen_space_radiance_cascades", "world_space_radiance_cascades") and no
# call interface of its own: its implementation surface is the lighting
# technique pass it attaches to the renderer (modules/renderer_vulkan/
# render_interface.hpp, an in-tree integration that is deliberately not a
# capability). Consumers demand one of the markers; nothing includes a header
# here, so no header-only include target exists.

# Excluded from the default build when the option is OFF: the target stays
# defined for consumers, but nothing builds the plugin binary unless an
# in-tree consumer actually links it (the runtime then simply does not see
# the plugin, and a demand for its capability fails startup precisely).
if(NOT ENGINE_BUILD_MODULE_RADIANCE_CASCADES)
    set_target_properties(engine_module_radiance_cascades PROPERTIES EXCLUDE_FROM_ALL ON)
endif()
